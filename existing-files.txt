export async function checkSupabaseHealth(): Promise<boolean> {
  try {
    const controller = new AbortController()
    const timeout = setTimeout(() => controller.abort(), 5000)
    
    const { data, error } = await supabase
      .from('profiles')
      .select('count', { count: 'exact', head: true })

export type SyncTable = 'logbook' | 'equipment' | 'profile'

export interface SyncQueueItem {
  id: string
  table: SyncTable
  operation: 'insert' | 'update' | 'delete'
  localId: string
  data: Record<string, any>
  timestamp: number
  retries: number
  error?: string
}

interface SyncConflict {
  item: SyncQueueItem
  error: string
  timestamp: number
  resolved: boolean
}

export interface SyncEngineState {
  isOnline: boolean
  isSyncing: boolean
  pendingCount: number
  lastSync: Date | null
  conflicts: SyncConflict[]
}

const STORAGE_KEY = 'dds_sync_queue'
const CONFLICTS_KEY = 'dds_sync_conflicts'
const ID_MAP_KEY = 'dds_id_map'
const LAST_SYNC_KEY = 'dds_last_sync'
const MAX_RETRIES = 3
const SYNC_INTERVAL_MS = 30000

class IdMap {
  private map: Map<string, string> = new Map()

  constructor() {
    this.load()
  }

  private load(): void {
    try {
      const stored = localStorage.getItem(ID_MAP_KEY)
      if (stored) {
        const obj = JSON.parse(stored)
        this.map = new Map(Object.entries(obj))
      }
    } catch { }
  }

  private save(): void {
    const obj = Object.fromEntries(this.map)
    localStorage.setItem(ID_MAP_KEY, JSON.stringify(obj))
  }

  set(localId: string, remoteId: string): void {
    this.map.set(localId, remoteId)
    this.save()
  }

  get(localId: string): string | undefined {
    return this.map.get(localId)
  }

  has(localId: string): boolean {
    return this.map.has(localId)
  }

  delete(localId: string): void {
    this.map.delete(localId)
    this.save()
  }
}

class SyncEngine {
  private queue: SyncQueueItem[] = []
  private isOnline: boolean = navigator.onLine
  private isSyncing: boolean = false
  private idMap: IdMap
  private listeners: Set<(state: SyncEngineState) => void> = new Set()
  private intervalId: number | null = null

  constructor() {
    this.idMap = new IdMap()
    this.loadQueue()
    this.setupNetworkListeners()
    this.startPeriodicSync()
  }

  async addToQueue(item: Omit<SyncQueueItem, 'id' | 'timestamp' | 'retries'>): Promise<void> {
    const fullItem: SyncQueueItem = {
      ...item,
      id: `op-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
      timestamp: Date.now(),
      retries: 0,
    }

    const existingIndex = this.queue.findIndex(
      q => q.localId === item.localId && q.table === item.table
    )
    if (existingIndex !== -1) {
      const existing = this.queue[existingIndex]
      
      if (item.operation === 'delete' && existing.operation === 'insert') {
        this.queue.splice(existingIndex, 1)
        this.saveQueue()
        this.notify()
        return
      }
      
      if (item.operation === 'update' && existing.operation === 'insert') {
        this.queue[existingIndex] = {
          ...existing,
          data: { ...existing.data, ...item.data },
          timestamp: Date.now(),
        }
        this.saveQueue()
        this.notify()
        return
      }
      
      if (item.operation === 'update' && existing.operation === 'update') {
        this.queue[existingIndex] = {
          ...existing,
          data: { ...existing.data, ...item.data },
          timestamp: Date.now(),
        }
        this.saveQueue()
        this.notify()
        return
      }
    }

    this.queue.push(fullItem)
    this.saveQueue()
    this.notify()

    if (this.isOnline && !this.isSyncing) {
      await this.processQueue()
    }
  }

  getState(): SyncEngineState {
    return {
      isOnline: this.isOnline,
      isSyncing: this.isSyncing,
      pendingCount: this.queue.length,
      lastSync: this.getLastSync(),
      conflicts: this.getConflicts(),
    }
  }

  subscribe(callback: (state: SyncEngineState) => void): () => void {
    this.listeners.add(callback)
    callback(this.getState())
    return () => this.listeners.delete(callback)
  }

  async forceSync(): Promise<void> {
    if (!this.isOnline) return
    await this.processQueue()
  }

  async resolveConflict(conflictId: string, resolution: 'local' | 'remote'): Promise<void> {
    const conflicts = this.getConflicts()
    const conflict = conflicts.find(c => c.item.id === conflictId)
    if (!conflict) return

    if (resolution === 'local') {
      await this.addToQueue({
        table: conflict.item.table,
        operation: conflict.item.operation,
        localId: conflict.item.localId,
        data: conflict.item.data,
      })
    }

    conflict.resolved = true
    this.saveConflicts(conflicts.filter(c => !c.resolved))
    this.notify()
  }

  private setupNetworkListeners(): void {
    window.addEventListener('online', () => {
      this.isOnline = true
      this.notify()
      this.processQueue()
    })
    window.addEventListener('offline', () => {
      this.isOnline = false
      this.notify()
    })
  }

  private startPeriodicSync(): void {
    this.intervalId = window.setInterval(() => {
      if (this.isOnline && this.queue.length > 0 && !this.isSyncing) {
        this.processQueue()
      }
    }, SYNC_INTERVAL_MS)
  }

  private async processQueue(): Promise<void> {
    if ('locks' in navigator) {
      try {
        await navigator.locks.request('dds-sync', { ifAvailable: true }, async (lock) => {
          if (!lock) return
          await this.doProcessQueue()
        })
      } catch {
        await this.doProcessQueue()
      }
    } else {
      await this.doProcessQueue()
    }
  }

  private async doProcessQueue(): Promise<void> {
    if (this.isSyncing || this.queue.length === 0) return
    this.isSyncing = true
    this.notify()

    const health = await checkSupabaseHealth()
    if (!health) {
      this.isSyncing = false
      this.notify()
      return
    }

    const userId = this.getCurrentUserId()
    if (!userId) {
      this.isSyncing = false
      this.notify()
      return
    }

    setSupabaseUserId(userId)

    const batch = this.queue.slice(0, 10)
    const remaining = this.queue.slice(10)

    try {
      const items = batch.map(item => ({
        table: item.table === 'logbook' ? 'logbook_entries' : item.table,
        operation: item.operation.toUpperCase(),
        local_id: item.localId,
        data: item.data,
      }))

      const { data, error } = await supabase.rpc('sync_batch', {
        p_user_id: userId,
        p_items: items,
      })

      if (error) throw error

      if (data?.results) {
        for (const result of data.results) {
          if (result.remote_id && result.local_id) {
            this.idMap.set(result.local_id, result.remote_id)
          }
        }
      }

      this.queue = remaining
      this.saveQueue()
      this.updateLastSync()

    } catch (error) {
      for (const item of batch) {
        item.retries++
        if (item.retries > MAX_RETRIES) {
          this.logConflict(item, error)
          const idx = this.queue.findIndex(q => q.id === item.id)
          if (idx !== -1) this.queue.splice(idx, 1)
        }
      }
      this.saveQueue()
    }

    this.isSyncing = false
    this.notify()

    if (this.queue.length > 0) {
      setTimeout(() => this.processQueue(), 1000)
    }
  }

  private getCurrentUserId(): string | null {
    try {
      const stored = localStorage.getItem('dds_current_user')
      if (stored) {
        const user = JSON.parse(stored)
        return user.codeId || user.id || null
      }
      return null
    } catch {
      return null
    }
  }

  private logConflict(item: SyncQueueItem, error: any): void {
    const conflicts = this.getConflicts()
    conflicts.push({
      item,
      error: error?.message || String(error),
      timestamp: Date.now(),
      resolved: false,
    })
    this.saveConflicts(conflicts)
  }

  private getConflicts(): SyncConflict[] {
    try {
      const stored = localStorage.getItem(CONFLICTS_KEY)
      return stored ? JSON.parse(stored) : []
    } catch {
      return []
    }
  }

  private saveConflicts(conflicts: SyncConflict[]): void {
    localStorage.setItem(CONFLICTS_KEY, JSON.stringify(conflicts))
  }

  private loadQueue(): void {
    try {
      const stored = localStorage.getItem(STORAGE_KEY)
      if (stored) this.queue = JSON.parse(stored)
    } catch { }
  }

  private saveQueue(): void {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.queue))
  }

  private getLastSync(): Date | null {
    try {
      const stored = localStorage.getItem(LAST_SYNC_KEY)
      return stored ? new Date(stored) : null
    } catch {
      return null
    }
  }

  private updateLastSync(): void {
    localStorage.setItem(LAST_SYNC_KEY, new Date().toISOString())
  }

  private notify(): void {
    const state = this.getState()
    this.listeners.forEach(cb => {
      try { cb(state) } catch { }
    })
  }

  destroy(): void {
    if (this.intervalId) clearInterval(this.intervalId)
  }
}

import { useEffect, useState, useCallback, useSyncExternalStore } from 'react'
import { syncEngine, type SyncEngineState } from '@/lib/syncEngine'
import { checkSupabaseHealth } from '@/lib/supabaseClient'

export function useSync() {
  const [isSupabaseConnected, setIsSupabaseConnected] = useState(false)
  const [supabaseError, setSupabaseError] = useState<string | null>(null)

  const syncState = useSyncExternalStore(
    syncEngine.subscribe.bind(syncEngine),
    () => syncEngine.getState(),
    () => syncEngine.getState()
  )

  useEffect(() => {
    const checkConnection = async () => {
      try {
        const health = await checkSupabaseHealth()
        setIsSupabaseConnected(health)
        setSupabaseError(null)
      } catch (err) {
        setIsSupabaseConnected(false)
        setSupabaseError(err instanceof Error ? err.message : 'Unknown error')
      }
    }

    checkConnection()
    const interval = setInterval(checkConnection, 30000)

    return () => clearInterval(interval)
  }, [])

  const forceSync = useCallback(() => {
    syncEngine.forceSync()
  }, [])

  const resolveConflict = useCallback((conflictId: string, resolution: 'local' | 'remote') => {
    syncEngine.resolveConflict(conflictId, resolution)
  }, [])

  return {
    ...syncState,
    isSupabaseConnected,
    supabaseError,
    canSync: syncState.isOnline && isSupabaseConnected,
    forceSync,
    resolveConflict,
  }
}

import { useState, useEffect, useCallback } from 'react';
import { getUserData, setUserData } from '@/lib/userStorage';

export interface LogEntry {
  id: string; dive_number: string; date: string; location: string; site: string;
  entry_time: string; exit_time: string; bottom_time: string; max_depth: string;
  avg_depth: string; psi_start: string; psi_end: string; tank_size: string;
  tank_type: string; dive_type: string; buddy: string; comments: string; diver_name: string;
}

const BASE_KEY = 'index_logbook_v4';

function loadLocal(): LogEntry[] { return getUserData<LogEntry[]>(BASE_KEY, []); }
function saveLocal(entries: LogEntry[]) { setUserData(BASE_KEY, entries); }

function migrateOldData(): LogEntry[] {
  const oldKey = 'index_logbook_v4';
  try {
    const raw = localStorage.getItem(oldKey);
    if (raw) {
      const parsed = JSON.parse(raw);
      if (Array.isArray(parsed) && parsed.length > 0) {
        const demoKey = `${BASE_KEY}_demo`;
        const existing = JSON.parse(localStorage.getItem(demoKey) || '[]');
        if (existing.length === 0) localStorage.setItem(demoKey, JSON.stringify(parsed));
        return parsed;
      }
    }
  } catch { }
  return [];
}

export function useLogbookStorage() {
  const [entries, setEntries] = useState<LogEntry[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [lastError, setLastError] = useState<string | null>(null);

  useEffect(() => {
    const migrated = migrateOldData();
    if (migrated.length > 0) setEntries(migrated);
    else setEntries(loadLocal());
    setIsLoading(false);
  }, []);

  useEffect(() => { if (!isLoading) saveLocal(entries); }, [entries, isLoading]);

  const addEntry = useCallback((entry: Omit<LogEntry, 'id'>) => {
    const newEntry: LogEntry = { ...entry, id: `log-${Date.now()}-${Math.random().toString(36).substr(2, 9)}` };
    setEntries(prev => [...prev, newEntry]);
  }, []);

  const saveEntry = addEntry;

  const updateEntry = useCallback((id: string, updates: Partial<LogEntry>) => {
    setEntries(prev => prev.map(entry => entry.id === id ? { ...entry, ...updates } : entry));
  }, []);

  const deleteEntry = useCallback((id: string) => {
    setEntries(prev => prev.filter(entry => entry.id !== id));
  }, []);

  const getEntryById = useCallback((id: string) => entries.find(entry => entry.id === id), [entries]);

  const syncWithSupabase = useCallback(async () => {
    try {
      setLastError(null);
      return { success: true, synced: 0 };
    } catch (error: any) {
      setLastError(error.message || 'Error de sync');
      return { success: false, error: error.message };
    }
  }, []);

  const syncNow = syncWithSupabase;

  return { entries, isLoading, lastError, addEntry, saveEntry, updateEntry, deleteEntry, getEntryById, syncWithSupabase, syncNow };
}

export interface EquipmentAlert {
  id: string; itemId: string; itemName: string;
  type: 'critical' | 'warning'; message: string; action?: string;
}

const BASE_KEY = 'divetools_equipment_v2';

function loadEquipment(): EquipmentItem[] {
  return getUserData<EquipmentItem[]>(BASE_KEY, []);
}

function saveEquipment(items: EquipmentItem[]) {
  setUserData(BASE_KEY, items);
}

function migrateOldData(): EquipmentItem[] {
  const keys = Object.keys(localStorage).filter(k => k.startsWith('divetools_equipment_') && !k.includes('_code-'));
  let allItems: EquipmentItem[] = [];
  for (const key of keys) {
    try {
      const raw = localStorage.getItem(key);
      if (raw) {
        const parsed = JSON.parse(raw);
        if (Array.isArray(parsed)) allItems = [...allItems, ...parsed];
      }
    } catch { }
  }
  if (allItems.length > 0) {
    const demoKey = `${BASE_KEY}_demo`;
    const existing = JSON.parse(localStorage.getItem(demoKey) || '[]');
    if (existing.length === 0) localStorage.setItem(demoKey, JSON.stringify(allItems));
    return allItems;
  }
  return [];
}

export function getAllAlerts(items: EquipmentItem[]): EquipmentAlert[] {
  const alerts: EquipmentAlert[] = [];
  const now = new Date();
  for (const item of items) {
    if (!item.alertEnabled) continue;
    if (item.lastServiceDate && item.serviceIntervalMonths) {
      const lastService = new Date(item.lastServiceDate);
      const nextService = new Date(lastService);
      nextService.setMonth(nextService.getMonth() + item.serviceIntervalMonths);
      if (now >= nextService) {
        alerts.push({ id: `svc-${item.id}`, itemId: item.id, itemName: `${item.brand} ${item.model}`, type: 'critical', message: `Mantencion vencida (${item.serviceIntervalMonths} meses)` });
      } else {
        const warningDate = new Date(nextService);
        warningDate.setDate(warningDate.getDate() - 30);
        if (now >= warningDate) {
          alerts.push({ id: `svc-w-${item.id}`, itemId: item.id, itemName: `${item.brand} ${item.model}`, type: 'warning', message: `Mantencion proxima (${item.serviceIntervalMonths} meses)` });
        }
      }
    }
    if (item.totalDives && item.serviceIntervalDives && item.totalDives >= item.serviceIntervalDives) {
      alerts.push({ id: `div-${item.id}`, itemId: item.id, itemName: `${item.brand} ${item.model}`, type: 'critical', message: `Mantencion por buceos (${item.totalDives}/${item.serviceIntervalDives})` });
    }
    if (item.lastPhDate && item.phIntervalYears) {
      const lastPh = new Date(item.lastPhDate);
      const nextPh = new Date(lastPh);
      nextPh.setFullYear(nextPh.getFullYear() + item.phIntervalYears);
      if (now >= nextPh) {
        alerts.push({ id: `ph-${item.id}`, itemId: item.id, itemName: `${item.brand} ${item.model}`, type: 'critical', message: `PH vencido (${item.phIntervalYears} anos)` });
      }
    }
    if (item.lastVisualDate && item.visualIntervalMonths) {
      const lastVisual = new Date(item.lastVisualDate);
      const nextVisual = new Date(lastVisual);
      nextVisual.setMonth(nextVisual.getMonth() + item.visualIntervalMonths);
      if (now >= nextVisual) {
        alerts.push({ id: `vis-${item.id}`, itemId: item.id, itemName: `${item.brand} ${item.model}`, type: 'critical', message: `Visual vencido (${item.visualIntervalMonths} meses)` });
      }
    }
  }
  return alerts;
}

export function useEquipmentStore() {
  const [items, setItems] = useState<EquipmentItem[]>(() => {
    const migrated = migrateOldData();
    if (migrated.length > 0) return migrated;
    return loadEquipment();
  });

  const addItem = useCallback((item: Omit<EquipmentItem, 'id'>) => {
    const newItem: EquipmentItem = { ...item, id: `eq-${Date.now()}-${Math.random().toString(36).substr(2, 9)}` };
    const updated = [...items, newItem];
    setItems(updated);
    saveEquipment(updated);
  }, [items]);

  const updateItem = useCallback((id: string, updates: Partial<EquipmentItem>) => {
    const updated = items.map(item => item.id === id ? { ...item, ...updates } : item);
    setItems(updated);
    saveEquipment(updated);
  }, [items]);

  const deleteItem = useCallback((id: string) => {
    const updated = items.filter(item => item.id !== id);
    setItems(updated);
    saveEquipment(updated);
  }, [items]);

  const getAlerts = useCallback(() => getAllAlerts(items), [items]);

  return { items, addItem, updateItem, deleteItem, getAlerts };
}

          </div>
        )}
      </div>
    </header>
  );
}



// src/lib/userStorage.ts
// Helper para aislamiento de datos por usuario (codeId)

import { useSessionStore } from '@/stores/useSessionStore';

export function getCurrentUserId(): string {
  try {
    const token = useSessionStore.getState().token;
    return token?.codeId || 'demo';
  } catch {
    return 'demo';
  }
}

export function getUserStorageKey(baseKey: string): string {
  const userId = getCurrentUserId();
  return `${baseKey}_${userId}`;
}

export function getUserData<T>(baseKey: string, defaultValue: T): T {
  try {
    const key = getUserStorageKey(baseKey);
    const raw = localStorage.getItem(key);
    if (raw) return JSON.parse(raw) as T;
  } catch { }
  return defaultValue;
}

export function setUserData<T>(baseKey: string, data: T): void {
  try {
    const key = getUserStorageKey(baseKey);
    localStorage.setItem(key, JSON.stringify(data));
  } catch { }
}

export function clearUserData(baseKey: string): void {
  try {
    const key = getUserStorageKey(baseKey);
    localStorage.removeItem(key);
  } catch { }
}

export function listUserStorageKeys(): string[] {
  const userId = getCurrentUserId();
  return Object.keys(localStorage).filter(k => k.endsWith(`_${userId}`));
}
